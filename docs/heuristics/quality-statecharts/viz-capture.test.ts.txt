/** Find a `machines` prop in a story's rendered element tree. */
function findMachines(element: unknown): Array<{ id: string; machine: unknown }> {
  if (!element || typeof element !== 'object') return [];
  const props = (element as { props?: Record<string, unknown> }).props;
  if (!props) return [];
  if (Array.isArray(props.machines)) return props.machines as never;
  const children = Array.isArray(props.children)
    ? props.children
    : [props.children];
  return children.flatMap(findMachines);
}

// --- sizing -----------------------------------------------------------------
type Measure = (text: string, font: string, letterSpacing?: number) => number;
const hasSize = (e: { width?: number; height?: number }) =>
  (e.width ?? 0) > 0 && (e.height ?? 0) > 0;
const stripParams = <T extends { params?: unknown }>(effects: T[] | undefined) =>
  (effects ?? []).map(({ params: _params, ...rest }) => rest as T);

function sanitizeNode(node: XGraphNode): XGraphNode {
  const seen = new Set<string>();
  return {
    ...node,
    data: {
      ...node.data,
      description: undefined,
      meta: null,
      assets: [],
      output: undefined,
      invokes: [],
      history: false,
      type: node.data.type === 'final' ? 'final' : 'normal',
      entry: stripParams(node.data.entry),
      exit: stripParams(node.data.exit),
      tags: (node.data.tags ?? [])
        .map((tag) => ({ ...tag, name: tag.name.trim() }))
        .filter((tag) => {
          const key = tag.name.toLocaleLowerCase();
          if (!tag.name || seen.has(key)) return false;
          seen.add(key);
          return true;
        }),
    },
  } as XGraphNode;
}

function eventLabel(eventType: string): string {
  if (!eventType) return 'always';
  const after = /^xstate\.after\.([^.]+)/.exec(eventType);
  if (after) return `after ${after[1]}`;
  if (eventType.startsWith('xstate.done.')) return 'done';
  if (eventType.startsWith('xstate.error.')) return 'error';
  if (eventType.startsWith('xstate.')) return eventType.slice(7);
  if (eventType === '*') return 'any event';
  return eventType;
}

function sanitizeEdge(edge: XGraphEdge, keep: 'guard' | 'actions'): XGraphEdge {
  const data = edge.data;
  return {
    ...edge,
    data: {
      ...data,
      eventType: eventLabel(data.eventType ?? ''),
      description: undefined,
      meta: undefined,
      matches: undefined,
      context: undefined,
      assets: [],
      guard: keep === 'guard' && data.guard ? { ...data.guard, params: undefined } : undefined,
      actions: keep === 'actions' ? stripParams(data.actions) : [],
    },
  } as XGraphEdge;
}

/**
 * Text-metric sizes as the editor estimates them; entities the estimator
 * defers to the DOM get the estimate of a sanitized copy plus fixed extras.
 */
function sizeGraph(graph: XGraph, measure: Measure): XGraph {
  const out = structuredClone(graph);
  const parents = new Set(out.nodes.map((node) => node.parentId));
  const one = (g: XGraph, id: string) => estimateEntitySizes(g, new Set([id]), measure)?.[0]?.size;
  for (const node of out.nodes) {
    if (hasSize(node)) continue;
    let size = one(out, node.id);
    if (!size) {
      const clean = sanitizeNode(node);
      size = one({ ...out, nodes: [clean], edges: [] }, node.id);
      if (size) {
        const invokes = node.data.invokes?.length ?? 0;
        size = {
          width: Math.max(size.width, invokes ? 200 : 0, parents.has(node.id) ? 240 : 0),
          height:
            size.height +
            (node.data.description ? 40 : 0) +
            (invokes ? 40 + 26 * invokes : 0) +
            (parents.has(node.id) ? 80 : 0),
        };
      }
    }
    if (size) Object.assign(node, size);
  }
  for (const edge of out.edges) {
    if (hasSize(edge)) continue;
    let size = one(out, edge.id);
    if (!size) {
      const g = sanitizeEdge(edge, 'guard');
      const a = sanitizeEdge(edge, 'actions');
      const gs = one({ ...out, edges: [g] }, edge.id);
      const as = one({ ...out, edges: [a] }, edge.id);
      if (gs && as)
        size = edge.data.guard && edge.data.actions?.length
          ? { width: Math.max(gs.width, as.width), height: gs.height + as.height - 40 }
          : edge.data.guard
            ? gs
            : as;
    }
    if (size) Object.assign(edge, size);
  }
  return out;
}

async function browserMeasure(texts: Set<string>): Promise<Map<string, number>> {
  const fontFile = path.resolve(
    'node_modules/@fontsource-variable/inter/files/inter-latin-wght-normal.woff2',
  );
  const browser = await chromium.launch();
  try {
    const page = await browser.newPage();
    await page.setContent('<html><body></body></html>');
    const base64 = fs.readFileSync(fontFile).toString('base64');
    const keys = [...texts];
    const widths = await page.evaluate(
      async ({ base64, keys }) => {
        const face = new FontFace('Inter Variable', `url(data:font/woff2;base64,${base64})`, {
          weight: '100 900',
        });
        await face.load();
        document.fonts.add(face);
        const context = document.createElement('canvas').getContext('2d')!;
        return keys.map((key) => {
          const [text, font, spacing] = JSON.parse(key) as [string, string, number];
          context.font = font;
          context.letterSpacing = `${spacing}px`;
          return Math.ceil(context.measureText(text).width * 64 - 1e-6) / 64;
        });
      },
      { base64, keys },
    );
    return new Map(keys.map((key, i) => [key, widths[i]!]));
  } finally {
    await browser.close();
  }
}

it('captures editor ELK inputs', { timeout: 3_600_000 }, async () => {
  const out = process.env.CAPTURE_OUT;
  if (!out) throw new Error('Set CAPTURE_OUT');
  void ESTIMATE_FONTS;
  const machines: Array<{ source: string; id: string; graph: XGraph }> = [];
  const failures: Array<{ source: string; id: string; error: string }> = [];
  for (const entry of entries()) {
    try {
      const graph = await entry.load();
      const normalized = createFlowStore(graph, { initialParentNodeId: null, mode: 'editing' })
        .getSnapshot().context.graph;
      machines.push({ source: entry.source, id: entry.id, graph: normalized });
    } catch (error) {
      failures.push({ source: entry.source, id: entry.id, error: String(error).slice(0, 300) });
    }
  }
  const texts = new Set<string>();
  const record: Measure = (text, font, spacing = 0) => {
    texts.add(JSON.stringify([text, font, spacing]));
    return 10;
  };
  for (const machine of machines) sizeGraph(machine.graph, record);
  const widths = await browserMeasure(texts);
  const measure: Measure = (text, font, spacing = 0) =>
    widths.get(JSON.stringify([text, font, spacing]))!;
  for (const machine of machines) machine.graph = sizeGraph(machine.graph, measure);

  const calls: Array<Record<string, unknown>> = [];
  let tag: Record<string, unknown> = {};
  let callIndex = 0;
  const original = ELK.prototype.layout;
  ELK.prototype.layout = function (graph: unknown, args?: unknown) {
    calls.push({
      ...tag,
      call: callIndex++,
      graph: structuredClone(graph),
      ...(args === undefined ? {} : { args: structuredClone(args) }),
    });
    return original.call(this, graph as never, args as never);
  } as typeof original;
  const layoutErrors: Array<Record<string, unknown>> = [];
  const summary: Array<Record<string, unknown>> = [];
  try {
    for (const machine of machines) {
      callIndex = 0;
      const before = calls.length;
      const scopes = [...getLayoutParentNodeIds(machine.graph)];
      for (const direction of ['DEFAULT', 'DOWN', 'RIGHT', 'UP', 'LEFT'] as const)
        for (const parentNodeId of scopes) {
          tag = { source: machine.source, machine: machine.id, direction, scope: parentNodeId };
          try {
            const flowGraph = graphToFlowGraph(machine.graph, {
              parentNodeId,
              depth: 1,
              inlineChildPadding: DEFAULT_INLINE_CHILD_PADDING,
            });
            await layoutGraph(
              machine.graph,
              flowGraph,
              parentNodeId,
              direction === 'DEFAULT' ? {} : { 'elk.direction': direction },
            );
          } catch (error) {
            layoutErrors.push({ ...tag, error: String(error).slice(0, 300) });
          }
        }
      summary.push({
        source: machine.source,
        id: machine.id,
        nodes: machine.graph.nodes.length,
        edges: machine.graph.edges.length,
        scopes: scopes.length,
        calls: calls.length - before,
      });
    }
  } finally {
    ELK.prototype.layout = original;
  }
  fs.writeFileSync(
    out,
    JSON.stringify({
      capturedAt: new Date().toISOString(),
      layoutPackage: '@statelyai/layout@0.4.3',
      machines: summary,
      conversionFailures: failures,
      layoutErrors,
      calls,
    }),
  );
  console.log(
    JSON.stringify({
      machines: machines.length,
      failures: failures.length,
      calls: calls.length,
      layoutErrors: layoutErrors.length,
    }),
  );
});
